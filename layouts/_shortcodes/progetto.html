{{- /* Scheda di un progetto della serie Progetti: link, tabella "In breve" e data di revisione. Uso, all'inizio del corpo del post: {{< progetto >}} Tutto ciò che cambia con le release vive nel blocco "project:" del front matter, non nel testo: aggiornare la pagina dopo una release significa cambiare "version", "status", "lastmod" e aggiungere una voce a "updates". Guida: docs/progetti.md. Controlli bloccanti (la build si ferma con un messaggio chiaro): - campi obbligatori mancanti; - "status" diverso da alpha, beta, stable; - "lastmod" diverso dalla data della voce più recente di "updates": la data mostrata al lettore e quella della cronologia devono coincidere; - copertina (cover.image, in static/) assente o non in 16:9. */ -}} {{- $page := .Page -}} {{- $p := $page.Params.project -}} {{- if not $p -}}{{- errorf "progetto: manca il blocco project: nel front matter (%s)" .Position -}}{{- end -}} {{- range slice "repo" "version" "status" "license" "what" "audience" "platforms" "updates" -}} {{- if not (index $p .) -}}{{- errorf "progetto: manca project.%s (%s)" . $.Position -}}{{- end -}} {{- end -}} {{- if not (in (slice "alpha" "beta" "stable") $p.status) -}} {{- errorf "progetto: project.status deve essere alpha, beta o stable, non %q (%s)" $p.status .Position -}} {{- end -}} {{- if not $page.Params.lastmod -}}{{- errorf "progetto: manca lastmod nel front matter (%s)" .Position -}}{{- end -}} {{- $latest := index (sort $p.updates "date" "desc") 0 -}} {{- $latestDate := time.AsTime $latest.date -}} {{- if ne ($latestDate.Format "2006-01-02") ($page.Lastmod.Format "2006-01-02") -}} {{- errorf "progetto: lastmod (%s) non coincide con la voce più recente di project.updates (%s) (%s)" ($page.Lastmod.Format "2006-01-02") ($latestDate.Format "2006-01-02") .Position -}} {{- end -}} {{- with $page.Params.cover.image -}} {{- $file := path.Join "static" . -}} {{- if not (fileExists $file) -}}{{- errorf "progetto: copertina %s non trovata (%s)" $file $.Position -}}{{- end -}} {{- $img := images.Config $file -}} {{- /* 16:9 con tolleranza di un pixel ogni cento, per gli arrotondamenti. */ -}} {{- if gt (math.Abs (sub (mul $img.Width 9) (mul $img.Height 16))) (div (mul $img.Width 9) 100) -}} {{- errorf "progetto: la copertina %s è %dx%d, deve essere in 16:9 (es. 1600x900) (%s)" $file $img.Width $img.Height $.Position -}} {{- end -}} {{- else -}} {{- errorf "progetto: manca cover.image nel front matter (%s)" .Position -}} {{- end -}} {{- $repo := printf "https://github.com/%s" $p.repo -}} {{- /* /releases/latest porta all'ultima release stabile; finché esistono solo pre-release quella pagina non c'è, quindi si punta all'elenco. */ -}} {{- $releases := cond (eq $p.status "stable") (printf "%s/releases/latest" $repo) (printf "%s/releases" $repo) -}} {{- /* "download" sostituisce il link alle release per i progetti che non ne pubblicano (per esempio un template da scaricare come ZIP). */ -}} {{- with $p.download }}{{ $releases = . }}{{ end -}}
